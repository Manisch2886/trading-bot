"""TB-103 B2: Dateizugriffe (audit 'open', ohne .py/.pyc) beim Import von paths OHNE Modus."""
import os, sys
ordner = sys.argv[1]
for v in ("TB_SELEKTIONSWURZEL", "TB_SELEKTIONSHASH", "TB_SELEKTIONSCOMMIT"):
    os.environ.pop(v, None)
zugriffe = []
def haken(e, a):
    if e == "open" and a and isinstance(a[0], (str, bytes)):
        p = os.fsdecode(a[0])
        if not p.endswith((".py", ".pyc")) and "__pycache__" not in p:
            zugriffe.append(p)
    if e in ("subprocess.Popen", "os.system", "os.listdir", "os.scandir"):
        zugriffe.append("%s %r" % (e, a[:1]))
sys.addaudithook(haken)
sys.path.insert(0, ordner)
import paths
print("%s: %d Zugriffe %s | DATA_DIR=%s CONFIG_DIR=%s" % (ordner, len(zugriffe), zugriffe, paths.DATA_DIR, paths.CONFIG_DIR))
